% CATALYTIC INVESTMENT: bullets on the left, feedback-loop figure from the Joule paper on the right
\begin{frame}{Catalytic Investment Mechanism}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \small
      \begin{itemize}\setlength{\itemsep}{0.5em}
        \item Each system is modelled for 2025, 2030, \ldots, 2050
        \item 2025 serves as a basic validation year
        \item 2030 to 2050 are first run as a \enquote{baseline} scenario without catalytic investment
        \item In \textit{catalysed} runs, advanced technologies are then pulled into the market, triggering technology learning and lower investment costs in subsequent years
      \end{itemize}
    \end{column}

\begin{frame}{Scenarios}
  \small
  \begin{itemize}\setlength{\itemsep}{0.6em}
    \item \textbf{Wind/solar acceptance limits:} land scarcity (e.g. Singapore), protected areas, \ldots
    \item \textbf{CDR breakthrough:} engineered CO\textsubscript{2} removal becomes cheap (e.g. 150 \$/tCO\textsubscript{2} by 2040)
    \item \textbf{Advanced baseload breakthrough:} SMRs or another firm clean technology see a rapid cost decline
    \item \textbf{Advanced geothermal breakthrough:} some regions gain access to high-temperature bedrock through advanced drilling
    \item \textbf{Solid oxide fuel cell breakthrough:} cheap backup power from hydrogen or from natural gas with carbon capture
  \end{itemize}
\end{frame}

\begin{frame}{Mathematics}

  Consider a total system cost minimisation problem\footnote{Assume the investment costs are annualised.}:

  \begin{equation*}
    \min_{g_{s,t}, G_s}\overbrace{\sum_s c_s G_s}^{\text{investment costs}} + \overbrace{\sum_{s,t} o_{s} g_{s,t}}^{\text{fuel costs}},   \qquad \text{s.t.}
  \end{equation*}
  \begin{align*}
    \sum_s g_{s,t} - d_t =  0 &\hspace{0.34cm}\perp \hspace{0.34cm} \lambda_t \tag{nodal energy balance}
  \end{align*}

  The dual variable $\lambda_t$ of the energy balance is the \alert{market price}.

\end{frame}
